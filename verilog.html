模块声明

模块是 Verilog 设计的基本单元，以 module 开头 endmodule 结尾

module.v
// 基本模块结构
module module_name #(
    parameter WIDTH = 8  // 可选参数
) (
    input  wire             clk,      // 输入端口
    input  wire             rst_n,    // 低电平复位
    input  wire [WIDTH-1:0]        data_in,  // 总线输入
    output reg  [WIDTH-1:0]        data_out, // 寄存器输出
    output wire            valid     // 组合逻辑输出
);

    // === 内部信号声明 ===
    wire [WIDTH-1:0] temp_wire;
    reg  [3:0]   cnt;

    // === 组合逻辑 ===
    assign temp_wire = data_in & 8'h0F;
    assign valid = |data_in;

    // === 时序逻辑 ===
    always @(posedge clk or negedge rst_n) begin
        if (!rst_n) begin
            data_out <= 0;
            cnt      <= 0;
        end else begin
            data_out <= temp_wire;
            cnt      <= cnt + 1;
        end
    end

endmodule

端口类型

模块与外部交互的通道，三种方向：输入、输出、双向

ports.v
// ========== 端口声明方式 ==========

// 方式一：模块声明时同时声明端口（推荐）
module example1 (
    input              a,           // 1bit 输入
    output             b,           // 1bit 输出
    inout              c            // 1bit 双向
);
endmodule

// 方式二：分开声明（旧式）
module example2(a, b, c);
    input  a;
    output b;
    inout  c;
endmodule

// ========== 带位宽的端口 ==========
module bus_example (
    input  wire [7:0]   data_in,   // 8bit 输入总线，高位在左
    output reg  [15:0]  data_out,  // 16bit 输出寄存器
    output wire [0:7]   rev_data   // 8bit 低位在左的小端序
);
endmodule

// ========== 端口信号类型 ==========
input  wire    sig1;   // 纯组合逻辑输入
input  reg     sig2;   // 过程块中的输入（不允许！）
output wire    sig3;   // 组合逻辑输出（常用 assign）
output reg     sig4;   // 时序/组合逻辑输出（常用 always）

数据类型

Verilog 中最常用的数据类型：wire（线网）、reg（寄存器）、parameter（参数）

datatype.v
// ========== wire（线网）==========
// 表示硬件连线，必须被驱动（assign 或由其它信号连接）
wire           single_bit;              // 1bit 线网
wire [3:0]   nibble;                  // 4bit 总线
wire [7:0]   byte = 8'hA5;      // 带初始值（综合时被忽略）

// ========== reg（寄存器）==========
// 表示存储单元，只能在 always/initial 块中赋值
reg            flag;
reg  [31:0]  counter;
reg  [0:7]   little_endian;     // 小端序定义

// ========== integer（整数）==========
// 至少32bit有符号整数，常用于循环计数和调试
integer i, j;
integer rom [0:255];               // 整数数组（ROM模型）

// ========== parameter（参数）==========
// 编译时常量，提高代码可复用性
parameter       CLK_FREQ   = 50_000_000;  // 50MHz系统
parameter       BIT_WIDTH = 8;
parameter       ADDR_W    = 10;
parameter       IDLE      = 2'b00,
                READ      = 2'b01,
                WRITE     = 2'b10;       // 状态机状态定义

// localparam：本地参数，不可通过例化覆盖
localparam     BURST_LEN = 16;

// ========== 数值表示 ==========
4'b1011    // 4bit 二进制: 1011
8'hA5      // 8bit 十六进制: 10100101
12'd1023  // 12bit 十进制: 001111111111
16'bx      // 16bit 不定态
8'bz       // 8bit 高阻态
'hFF       // 默认位宽（至少32bit）的十六进制
32'd100    // 显式32bit十进制

always 与 initial 过程块

always 块描述组合逻辑或时序逻辑；initial 块仅在仿真开始时执行一次

process.v
// ========== 时序逻辑 always 块 ==========
// 记得描述复位！同步复位或异步复位
always @(posedge clk) begin          // 同步复位
// always @(posedge clk or posedge rst) begin  // 异步复位（同时监听复位信号边沿）
    if (rst) begin
        q <= 1'b0;
    end else begin
        q <= d;
    end
end

// ========== 组合逻辑 always 块 ==========
// 敏感列表要完整！推荐使用 @* 自动敏感列表
always @* begin
    case (state)
        IDLE:  next = READ;
        READ:  next = WRITE;
        WRITE: next = IDLE;
        default: next = IDLE;
    endcase
end

// ========== 完整敏感列表写法 ==========
always @(a or b or sel) begin          // 组合逻辑
    if (sel)
        y = a;
    else
        y = b;
end

// ========== initial 块（仅仿真） ==========
initial begin
    clk = 0;
    rst = 1;
    #100 rst = 0;   // #延时，单位为仿真时间单位
end

// 产生时钟
always #(10) clk = !clk;  // 每10个时间单位翻转一次

连续赋值 assign

用于组合逻辑，右边信号变化时立即更新左边，always 块外使用

assign.v
// ========== 基本赋值 ==========
assign y = a & b;              // 与门
assign y = a | b;              // 或门
assign y = ~a;                 // 非门
assign y = a ^ b;              // 异或

// ========== 多位宽操作 ==========
assign sum  = a + b;             // 加法（会自动扩展位宽）
assign diff = a - b;             // 减法
assign y    = a >> 2;            // 逻辑右移（补0）
assign y    = a >>> 2;           // 算术右移（补符号位）
assign y    = a << 1;            // 左移（补0）

// ========== 位拼接 ==========
assign y = {a, b, c};              // 拼接：{bit2, bit1, bit0}
assign y = {2{a}};              // 重复：2'b aa
assign y = {a, 1'b0, b[3:1]};     // 混合拼接

// ========== 三元运算符（条件赋值） ==========
assign y = (sel) ? a : b;           // sel=1选a，sel=0选b，等效二选一mux
assign y = (a > b) ? a : b;       // 比较后取大者
assign overflow = (a[31] == b[31]) & (result[31] != a[31]);  // 溢出检测

运算符

算术、逻辑、位运算、关系和等价运算符一览

operators.txt
// ========== 算术运算符 ==========
+   // 加法  a + b
-   // 减法  a - b
*   // 乘法  a * b
/   // 除法（综合可能被优化为移位）  a / b
%   // 取模  a % b

// ========== 位运算符（按位操作，每位单独运算）==========
&   // 按位与  a & b
|   // 按位或  a | b
^   // 按位异或  a ^ b
~   // 按位取反  ~a
~^  // 按位同或（异或非）

// ========== 逻辑运算符（返回1bit真假）==========
&&  // 逻辑与  a && b   （结果：0或1）
||  // 逻辑或  a || b   （结果：0或1）
!   // 逻辑非  !a       （结果：0或1）

// ========== 关系运算符 ==========
<   // 小于
>   // 大于
<=  // 小于等于
>=  // 大于等于

// ========== 等价运算符 ==========
==  // 相等（x和z参与比较，结果可能为x）
!=  // 不等
=== // 绝对等于（x和z严格比较，仅仿真用）
!== // 绝对不等

// ========== 移位运算符 ==========
<<  // 逻辑左移  a << n  （低位补0）
>>  // 逻辑右移  a >> n  （高位补0）
<<< // 算术左移  a <<< n （低位补0，等同逻辑左移）
>>> // 算术右移  a >>> n （高位补符号位，有符号数用）

// ========== 归约运算符（一元，对所有位归约）==========
&   a   // 归约与：所有位相与，&a[3]&a[2]&a[1]&a[0]
~&  a   // 归约与非
|   a   // 归约或
~|  a   // 归约或非
^   a   // 归约异或（奇偶校验）
~^  a   // 归约同或

// ========== 拼接与重复 ==========
{ }   // 拼接 {a, b, c[1:0]}
{{ }} // 重复 {{n}{a}} 等价于 {a,a,...,a}(n个)

// ========== 条件运算符 ==========
?:  // 三元：cond ? true_val : false_val

流程控制语句

if-else、case、for、while 等过程块内使用的控制语句

control.v
// ========== if-else ==========
if (a > 8) begin
    y = 1;
end else if (a == 8) begin
    y = 0;
end else begin
    y = -1;
end

// ========== case / casez / casex ==========
// case  : 严格比较，x/z 参与比较
// casez : z 被当作不关心（? 可以代替）
// casex : x 和 z 都被当作不关心（极少用，不推荐）
always @* {
    case (sel)
        2'b00:   y = a;
        2'b01:   y = b;
        2'b1?:    y = c;     // casez: ? 表示不关心
        default:  y = 0;
    endcase
}

// ========== for 循环 ==========
integer i;
always @* {
    for (i = 0; i < 8; i = i+1) begin
        result[i] = a[i] & enable;
    end
}

// ========== while 循环 ==========
integer j;
initial begin
    j = 0;
    while (j < 16) begin
        rom[j] = j * 2;
        j = j + 1;
    end
end

// ========== repeat / forever ==========
initial begin
    repeat (5) #(10) data = data + 1;  // 重复5次，间隔10时间单位
end

// forever 会无限循环（需配合 disable 跳出，仅仿真用）
initial begin
    forever #(5) clk = !clk;
end

// ========== 块名与 disable ==========
break_block: begin : break_block
    forever begin
        if (cond) disable break_block;  // 跳出命名块
    end
end

模块例化

将模块实例化为硬件电路，模块间连接的桥梁

inst.v
// ========== 位置关联（按顺序）==========
my_module u0 (
    clk,
    rst_n,
    data_in,
    data_out
);

// ========== 名称关联（推荐，不易出错）==========
my_module #(
    .WIDTH(WIDTH),       // 参数传递
    .DEPTH(DEPTH)
) u1 (
    .clk(sys_clk),
    .rst_n(sys_rst),
    .data_in(din),
    .data_out(dout)
);

// ========== 悬空端口 ==========
// 未连接的端口自动悬空（综合优化）
my_module u2 (
    .clk(clk),
    .rst_n(),           // 悬空（不连接）
    .data_in(din),
    .data_out()            // 悬空
);

// ========== 未连接端口赋默认值 ==========
// 某些工具支持 .port_name() 留空，或直接不写

存储器（数组）

一维数组（存储器）和二维数组的表达方式

memory.v
// ========== 一维存储器（word数组）==========
reg [7:0] ram [0:255];  // 256个8bit字的存储器（RAM模型）
reg [31:0] rom [0:1023]; // 1024个32bit字的ROM

// ========== 读写操作 ==========
// 读（组合逻辑，读无需时钟）
wire [7:0] data_r = ram[addr];

// 写（时序逻辑）
always @(posedge clk) begin
    if (we) begin
        ram[addr] <= data_in;
    end
end

// ========== 二维数组（展开为矩阵）==========
reg [7:0] matrix [0:3][0:7];  // 4行8列，每格8bit
// matrix[2][5] 表示第2行第5列的8bit数据
matrix[i][j] <= value;        // 访问某行某列

// ========== bit/piece select（位选/段选）==========
data[3]           // 单独选1bit
data[7:4]         // 选 [7:4] 共4bit，高位在左
data[0+:4]        // 从0开始选4bit，等价 [3:0]
data[7-:8]        // 从7开始选8bit（变址），等价 [7:0]

任务（task）与函数（function）

task 可包含时序控制和多语句；function 仅实现组合逻辑，必须在单一时钟内完成

taskfunc.v
// ========== function（组合逻辑，自动推断）==========
// 必须有输入，至少一个输出，不能有时延 # 或阻塞 wait
function [7:0] parity;
    input [7:0] data;
    integer i;
    begin
        parity = 0;
        for (i=0; i<8; i=i+1)
            parity = parity ^ data[i];  // 异或归约求奇偶
    end
endfunction

// 调用
wire [7:0] p = parity(data_byte);

// ========== task（可含时序，更灵活）==========
task send_byte;
    input [7:0] data;
    integer i;
    begin
        for (i=0; i<8; i=i+1) begin
            tx_line <= data[i];    // 逐bit发送
            #(period);                        // 延时（task中可用）
        end
    end
endtask

// 调用
initial send_byte(8'hA5);

// ========== automatic task/function（仿真中常用）==========
function automatic [31:0] gcd;  // 可重入，每次调用独立栈
    input [31:0] a, b;
    begin
        while (a != b) begin
            if (a > b) a = a - b;
            else         b = b - a;
        end
        gcd = a;
    end
endfunction

编译指令

以反引号 ` 开头，在编译时生效，不加分号结尾

directive.v
// ========== `define — 文本宏 ==========
`define CLK_PERIOD 20              // 定义常量
`define DATA_W     8                // 总线位宽宏
`define MSG        "Error"         // 字符串宏
`define MAX(a, b) (a>b?a:b)      // 带参数的宏函数
// 使用：#`CLK_PERIOD   或   y = `MAX(x, z)
// 取消定义：`undef CLK_PERIOD

// ========== `include — 文件包含 ==========
`include "../../defines.v"     // 包含公共定义文件
`include <header.vh>        // 尖括号表示库搜索路径

// ========== `timescale — 时间尺度（仿真用）==========
`timescale 1ns/1ps              // 时间单位1ns，精度1ps
// 延时 #10 表示 10ns；精度决定四舍五入

// ========== 条件编译 ==========
`ifdef DEBUG
    always @(posedge clk) $display("pc=%h", pc);
`endif

`ifndef SAFE_MODE          // if not defined
    define SAFE_MODE           // 未定义则自动定义
`endif

// ========== 其他常用指令 ==========
`default_nettype wire          // 未声明信号默认wire（建议加，防止意外）
`celldefine / `endcelldefine   // 标记模块为库单元
`resetall                      // 重置所有编译指令为默认值
